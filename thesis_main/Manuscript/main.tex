% --- Float placement (APA 7: embedded figures sit at the TOP or BOTTOM of a
% shared page, never mid-text). LaTeX's stock float parameters are too strict
% and eject large figures onto near-blank pages; loosen them so a figure may
% fill most of a page top/bottom before spilling, and require very little text
% to accompany a float. This is what eliminates the wasted-whitespace pages. ---
\renewcommand{\topfraction}{0.9}      % a top float may occupy up to 90% of the page
\renewcommand{\bottomfraction}{0.9}   % same for bottom floats
\renewcommand{\textfraction}{0.08}    % as little as 8% text needed on a float page-share
\renewcommand{\floatpagefraction}{0.7}% only bump to a float-only page if it fills 70%+
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
% Default all figure/table floats to top/bottom/float-page (APA), overriding the
% class's htb default; explicit per-float specifiers still win where given.
\floatplacement{figure}{tbp}
\floatplacement{table}{tbp}
% --- Breathing room above/below every in-text [H] float. Ms. Kim asked for
% "space before" tables and figures throughout; float.sty emits \intextsep both
% above and below an [H] float, so widening it once spaces all of them uniformly
% (default is ~12pt, i.e. under one double-spaced line — too tight against text). ---
\setlength{\intextsep}{20pt plus 5pt minus 3pt}
